/* Lessons per subject, in chronological order (oldest first) as authored in
   the data files. The menu displays them newest-first.
   Shape: { id, label, topic, group, words: [{en,sk}], sides? } */
let lessons = { english: [], science: [] };

function lessonsFor(subject) { return lessons[subject] || []; }
function lessonById(subject, id) { return lessonsFor(subject).find(l => l.id === id) || null; }

/* Newest-first, grouped for display. Group order follows first appearance in
   the newest-first list, so the current month always sits on top. */
function lessonGroups(subject) {
  const out = [];
  for (const l of [...lessonsFor(subject)].reverse()) {
    const name = l.group || 'Lekcie';
    let g = out.find(x => x.name === name);
    if (!g) { g = { name, lessons: [] }; out.push(g); }
    g.lessons.push(l);
  }
  return out;
}

// Merged, de-duplicated word list for a selection of lessons.
function wordsForSelection(subject, ids) {
  const seen = new Set(); const out = [];
  for (const id of ids) {
    for (const p of (lessonById(subject, id)?.words || [])) {
      const k = `${p.en}|${p.sk}`;
      if (seen.has(k)) continue;
      seen.add(k); out.push(p);
    }
  }
  return out;
}

/* Makes a selection playable by the game engine. A single lesson plays under
   its own id (so the leaderboard/labels stay meaningful); several lessons get
   merged into one temporary combined unit. */
function unitForSelection(subject, ids) {
  if (ids.length === 1) return ids[0];
  const key = `__combo__${subject}`;
  trickyUnits[key] = wordsForSelection(subject, ids);
  UNIT_LABELS[key] = { left: 'EN', right: 'SK' };
  return key;
}

function selectionLabel(subject, ids) {
  if (!ids.length) return '';
  if (ids.length === 1) return lessonById(subject, ids[0])?.label || ids[0];
  return `${ids.length} lekcie`;
}

// Turns an English answer into the filename convention used under images/story/.
// "virtual reality headset" -> "virtual_reality_headset" (expects a matching .jpg)
function storySlug(en) {
  return en.trim().toLowerCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .replace(/[^a-z0-9\s-]/g, '')
    .replace(/\s+/g, '_');
}

function parseWords(src) {
  wordPool = [];
  for (const line of src.split('\n')) {
    const t = line.trim();
    if (!t || t.startsWith('#')) continue;
    const sep = t.indexOf('|');
    if (sep < 0) continue;
    const level = t.slice(0, sep).trim().toUpperCase();
    const word  = t.slice(sep + 1).trim().toLowerCase();
    if (level && word) wordPool.push({ level, word });
  }
}

function parseTranslationWords(src) {
  const pool = [];
  for (const line of src.split('\n')) {
    const t = line.trim();
    if (!t || t.startsWith('#')) continue;
    const sep = t.indexOf('|');
    if (sep < 0) continue;
    const en = t.slice(0, sep).trim().toLowerCase();
    const sk = t.slice(sep + 1).trim().toLowerCase();
    if (en && sk) pool.push({ en, sk });
  }
  return pool;
}

// Fetches up to 100 highest global scores, sorted descending. Calls
// cb(null) on any failure (offline, Firebase down, etc.) so callers can
// fall back to the local list without special-casing errors themselves.
function fetchGlobalLeaderboard(cb) {
  if (!fbDb) { cb(null); return; }
  fbDb.ref('leaderboard').orderByChild('score').limitToLast(100).once('value')
    .then(snap => {
      const arr = [];
      snap.forEach(child => arr.push(child.val()));
      arr.sort((a, b) => b.score - a.score);
      cb(arr);
    })
    .catch(err => { console.warn('Global leaderboard fetch failed:', err); cb(null); });
}

// Fire-and-forget — never blocks or breaks gameplay if it fails.

  if (!fbDb) return;
  try { fbDb.ref('leaderboard').push({ ...entry, ts: Date.now() }); }
  catch (err) { console.warn('Global leaderboard write failed:', err); }
}

function isTopTen(score) {
  const lb = getLeaderboard();
  return lb.length < 10 || score > lb[9].score;

function addToLeaderboard(name, score, startLevel, reachedLevel, kid, mode) {
  const entry = {
    name: (name.trim() || 'Anonymous').slice(0, 16),
    score, startLevel, reachedLevel, kid: !!kid,
    mode: mode || 'legacy',
    date: new Date().toLocaleDateString(),
  };
  const lb = getLeaderboard();
  lb.push(entry);
  lb.sort((a, b) => b.score - a.score);
  if (lb.length > LB_MAX) lb.splice(LB_MAX);
  saveLeaderboard(lb);
  pushToGlobalLeaderboard(entry);
}

/* ── Diacritic-insensitive comparison ──────────────────────────────────────── */
function normalize(s) {
  return s.normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().replace(/\s+/g, '');
}

// Capitalizes the first letter of every word — used only for the "what you
// missed" recap text so e.g. "the united kingdom" reads as "The United Kingdom".
function capWords(s) {
  return s.replace(/\b\p{L}/gu, c => c.toUpperCase());
}

/* ════════════════════════════════════════════════════════════════════════════
   PRELOAD SCENE — tries to load a picture (images/story/<slug>.jpg) for every
   tricky word. Whatever loads successfully becomes playable in Story Mode;
   whatever is missing is silently skipped — no manifest file needed, just
   drop a correctly-named .jpg into images/story/ and reload.
════════════════════════════════════════════════════════════════════════════ */
class PreloadScene extends Phaser.Scene {
  constructor() { super('Preload'); }

  preload() {
    this.add.rectangle(W / 2, H / 2, W, H, 0x0d0d1a);
    drawGrid(this, 0x141428, 0.5);
    this.add.text(W / 2, H / 2 - 46, 'WORD DROP', {
      fontSize: '46px', fill: '#00ccff',
      fontFamily: '"Arial Black", Arial, sans-serif',
      stroke: '#004488', strokeThickness: 8,
    }).setOrigin(0.5);

    const barBg = this.add.rectangle(W / 2, H / 2 + 20, 320, 18, 0x1c2a3a).setStrokeStyle(2, 0x334455);
    const barFg = this.add.rectangle(W / 2 - 158, H / 2 + 20, 4, 14, 0x00ccff).setOrigin(0, 0.5);
    this.add.text(W / 2, H / 2 + 48, 'Loading pictures…', {
      fontSize: '13px', fill: '#7788aa', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5);

    // Build one candidate per unique (unit, word) — de-duped by slug within
    // each unit. Each candidate gets a plain numeric index, which becomes
    // the image key; simpler and collision-proof than encoding the slug
    // itself in the key (units can't collide with each other this way).
    this._storyCandidates = [];
    Object.entries(trickyUnits).forEach(([unitName, pairs]) => {
      const seenInUnit = new Set();
      pairs.forEach(p => {
        const slug = storySlug(p.en);
        if (seenInUnit.has(slug)) return;
        seenInUnit.add(slug);
        this._storyCandidates.push({ unitName, en: p.en, sk: p.sk, slug });
      });
    });

    // Try both extensions per word — AI image tools mostly export .png, but
    // .jpg should work too. Whichever exists gets used (if both exist, the
    // one that finishes loading first wins; the addedIdx guard below stops
    // a word from being added twice).
    const EXTS = ['jpg', 'png'];
    this._storyCandidates.forEach((c, i) => {
      EXTS.forEach(ext => this.load.image(`story_${i}__${ext}`, `images/story/${c.slug}.${ext}`));
    });

    const addedIdx = new Set();
    this.load.on('progress', v => { barFg.width = Math.max(4, 316 * v); });
    this.load.on('filecomplete', key => {
      const m = /^story_(\d+)__(jpg|png)$/.exec(key);
      if (!m) return;
      const idx = Number(m[1]);
      if (addedIdx.has(idx)) return; // already picked up via the other extension
      const c = this._storyCandidates[idx];
      if (!c) return;
      addedIdx.add(idx);
      (storyUnits[c.unitName] ||= []).push({ en: c.en, sk: c.sk, imageKey: key });
    });
    this.load.on('loaderror', () => { /* missing picture — expected, ignore */ });
  }

  create() {
    this.scene.start('Menu');
  }
}

  }).setOrigin(0.5);
  const desc = scene.add.text(x, y + 28, '', {
    fontSize: '12px', fill: LEVEL_COLORS[3], fontFamily: 'Arial, sans-serif',
  }).setOrigin(0.5);

  const refresh = () => {
    const lv = MENU_STATE.level;
    num.setText(`${lv}`).setStyle({ fill: LEVEL_COLORS[lv] });
    desc.setText(DIFF_LABELS[lv]).setStyle({ fill: LEVEL_COLORS[lv] });
    if (onChange) onChange();
  };
  const step = d => { MENU_STATE.level = Phaser.Math.Clamp(MENU_STATE.level + d, 0, 9); refresh(); };

  [[lArr, -1], [rArr, 1]].forEach(([arr, d]) => {
    arr.on('pointerover', () => arr.setStyle({ fill: '#aabbcc' }));

  scene.input.keyboard.on('keydown', e => {
    if (e.key >= '0' && e.key <= '9') { MENU_STATE.level = parseInt(e.key, 10); refresh(); }
  });

  refresh();
  return { refresh };
}

    this.add.text(VIEW.x + 8, 88, 'LEKCIA', {
      fontSize: '11px', fill: '#667788',
      fontFamily: '"Arial Black", Arial, sans-serif', letterSpacing: 2,
    }).setOrigin(0, 1);

    MODES.forEach(m => {
      const x = this.colX[m.id];
      this.add.text(x, 70, m.icon, { fontSize: '15px' }).setOrigin(0.5);
      this.add.text(x, 86, m.label, {
        fontSize: '10px', fill: m.soon ? '#8a7a55' : '#aabbcc',
        fontFamily: '"Arial Black", Arial, sans-serif', align: 'center',
        wordWrap: { width: 84 },
      }).setOrigin(0.5, 0.5);
    });
    this.add.rectangle(W / 2, VIEW.y - 4, 720, 1, 0x2a3c50);

    /* ── Scrollable list ── */
    this.listCt = this.add.container(0, 0);
    const maskG = this.make.graphics({ add: false });
    maskG.fillRect(VIEW.x, VIEW.y, VIEW.w, VIEW.h);
    this.listCt.setMask(maskG.createGeometryMask());

    this.scrollY = 0;
    this.contentH = 0;
    this.scrollBar = this.add.rectangle(VIEW.x + VIEW.w - 3, VIEW.y, 4, 40, 0x33475e)
      .setOrigin(0.5, 0);

    this.buildList();

    // Wheel + drag scrolling
    this.input.on('wheel', (p, go, dx, dy) => this.scrollBy(dy * 0.5));
    this.input.on('pointermove', p => {
      if (!p.isDown) return;
      if (p.y < VIEW.y || p.y > VIEW.y + VIEW.h) return;
      this.scrollBy(-(p.y - p.prevPosition.y));
    });

    /* ── Bottom bar: difficulty + selection + START ──
       Order matters: the texts have to exist before difficultyPicker() runs,
       because its initial refresh calls back into refreshInfo(). */
    this.add.rectangle(W / 2, 424, 720, 1, 0x2a3c50);

  /* Rebuilds every row. Cheap enough to just redo on any change (selection,
     collapsing a group) — keeps the state handling in one place. */
  buildList() {
    this.listCt.removeAll(true);
    const VIEW = this.VIEW;
    let y = VIEW.y + 6;
    const groups = lessonGroups(this.subject);

    if (!groups.length) {
      this.listCt.add(this.add.text(W / 2, VIEW.y + 80,
        `Žiadne lekcie.\nPridaj prvú do ${this.cfg.source} a stlač F5.`, {
          fontSize: '15px', fill: '#445566', fontFamily: 'Arial, sans-serif', align: 'center',
        }).setOrigin(0.5));
      this.contentH = 0;
      this.applyScroll();
      return;
    }

    groups.forEach(g => {
      const open = !this.collapsed[g.name];

      // Group header (click to collapse/expand)
      const hBg = this.add.rectangle(W / 2, y + 13, VIEW.w - 16, 26, 0x16202c)
        .setInteractive({ useHandCursor: true });
      const hTxt = this.add.text(VIEW.x + 16, y + 13, `${open ? '▾' : '▸'}  ${g.name}`, {
        fontSize: '12px', fill: '#8899aa', fontFamily: '"Arial Black", Arial, sans-serif',
      }).setOrigin(0, 0.5);
      const hCnt = this.add.text(VIEW.x + VIEW.w - 24, y + 13,
        `${g.lessons.length} ${g.lessons.length === 1 ? 'lekcia' : 'lekcie'}`, {
          fontSize: '11px', fill: '#556677', fontFamily: 'Arial, sans-serif',
        }).setOrigin(1, 0.5);
      hBg.on('pointerdown', p => {
        if (!this.inView(p)) return;   // row scrolled out of sight (mask hides
        if (!open) {                   // the pixels, but not the hit area)
          delete this.collapsed[g.name];
        } else {
          this.collapsed[g.name] = true;
        }
        this.buildList();
      });
      this.listCt.add([hBg, hTxt, hCnt]);
      y += 30;

      if (!open) { y += 4; return; }

      g.lessons.forEach(l => {
        const rowY = y + 18;
        this.listCt.add(this.add.rectangle(W / 2, y, VIEW.w - 16, 1, 0x223040));
        this.listCt.add(this.add.text(VIEW.x + 16, rowY - 7, l.label, {
          fontSize: '15px', fill: '#dde6ee', fontFamily: 'Arial, sans-serif',
        }).setOrigin(0, 0.5));
        this.listCt.add(this.add.text(VIEW.x + 16, rowY + 10,
          `${l.topic ? l.topic + '  ·  ' : ''}${l.words.length} slovíčok`, {
            fontSize: '11px', fill: '#667788', fontFamily: 'Arial, sans-serif',
          }).setOrigin(0, 0.5));

        MODES.forEach(m => this.addCheckbox(this.colX[m.id], rowY, l.id, m.id));
        y += 38;
      });
      y += 6;
    });

    this.contentH = y - VIEW.y;
    this.applyScroll();
  }

  addCheckbox(x, y, lessonId, modeId) {
    const on = this.sel.mode === modeId && this.sel.ids.includes(lessonId);
    const box = this.add.rectangle(x, y, 22, 22, on ? this.cfg.colorHex : 0x16202c)
      .setStrokeStyle(2, on ? this.cfg.colorHex : 0x33475e)
      .setInteractive({ useHandCursor: true });
    const tick = this.add.text(x, y - 1, on ? '✓' : '', {
      fontSize: '14px', fill: '#0d1520', fontFamily: '"Arial Black", Arial',
    }).setOrigin(0.5);
    this.listCt.add([box, tick]);

    // Tap = toggle, but ignore taps that were really a scroll drag.
    let downY = null;
    box.on('pointerdown', p => { downY = this.inView(p) ? p.y : null; });
    box.on('pointerup', p => {
      // A tap toggles; a drag was meant as scrolling, so it's ignored. Rows
      // scrolled out of the viewport are ignored too — the mask hides their
      // pixels but their hit area would still be clickable.
      if (downY === null || !this.inView(p) || Math.abs(p.y - downY) > 8) { downY = null; return; }
      downY = null;
      this.toggle(lessonId, modeId);
    });
    box.on('pointerover', () => { if (!on) box.setStrokeStyle(2, this.cfg.colorHex); });
    box.on('pointerout',  () => { if (!on) box.setStrokeStyle(2, 0x33475e); });
  }

  /* Selection rules: one mode at a time (a mix of "Study week 3" + "Drop
     week 4" has no meaning), and up to MAX_SELECTED_LESSONS lessons inside
     that mode. Raise that constant to allow several weeks at once. */
  toggle(lessonId, modeId) {
    const s = this.sel;
    if (s.mode !== modeId) { s.mode = modeId; s.ids = [lessonId]; }
    else if (s.ids.includes(lessonId)) {
      s.ids = s.ids.filter(i => i !== lessonId);
      if (!s.ids.length) s.mode = null;
    } else if (s.ids.length < MAX_SELECTED_LESSONS) {
      s.ids.push(lessonId);
    } else {
      s.ids = [lessonId];   // single-select behaviour: replace
    }
    this.buildList();
    this.refreshInfo();
  }

  // Is the pointer inside the scrollable list area?
  inView(p) {
    return p.y >= this.VIEW.y && p.y <= this.VIEW.y + this.VIEW.h;
  }

  scrollBy(dy) {
    const max = Math.max(0, this.contentH - this.VIEW.h);
    this.scrollY = Phaser.Math.Clamp(this.scrollY + dy, 0, max);
    this.applyScroll();
  }

  applyScroll() {
    const max = Math.max(0, this.contentH - this.VIEW.h);
    this.scrollY = Phaser.Math.Clamp(this.scrollY, 0, max);
    this.listCt.y = -this.scrollY;

    // Scrollbar: hidden when everything fits
    if (max <= 0) { this.scrollBar.setVisible(false); return; }
    const frac = this.VIEW.h / this.contentH;
    const barH = Math.max(28, this.VIEW.h * frac);
    this.scrollBar.setVisible(true).setSize(4, barH);
    this.scrollBar.y = this.VIEW.y + (this.VIEW.h - barH) * (this.scrollY / max);
  }

  refreshInfo() {
    if (!this.selTxt) return;   // bottom bar not built yet (see create())
    const s = this.sel;
    if (!s.mode || !s.ids.length) {
      this.selTxt.setText('Nič nie je vybrané');
      this.infoTxt.setText('Zaškrtni políčko v riadku lekcie a v stĺpci módu.');
      this.startBtn.bg.setFillStyle(0x223244).disableInteractive();
      this.startBtn.txt.setStyle({ fill: '#556677' });
      return;
    }
    const m = modeById(s.mode);
    const n = wordsForSelection(this.subject, s.ids).length;
    this.selTxt.setText(`${selectionLabel(this.subject, s.ids)}  ·  ${m.label}`);
    this.infoTxt.setText(`${m.desc}  (${n} slovíčok)`);
    this.startBtn.bg.setFillStyle(0x0066bb).setInteractive({ useHandCursor: true });
    this.startBtn.txt.setStyle({ fill: '#ffffff' });
  }

  start() {
    const s = this.sel;
    if (!s.mode || !s.ids.length) return;
    const m    = modeById(s.mode);
    const back = { scene: 'Subject', data: { subject: this.subject } };

    if (m.scene === 'Study') {
      this.scene.start('Study', { subject: this.subject, ids: [...s.ids], back });
    } else if (m.scene === 'Train') {
      this.scene.start('Train', { subject: this.subject, ids: [...s.ids], back });
    } else {
      this.scene.start('Game', {
        level:  MENU_STATE.level,
        mode:   m.mechanic,
        unit:   unitForSelection(this.subject, s.ids),
        lbMode: this.cfg.lbMode,
        back,
      });
    }
  }
}

  }

/* ════════════════════════════════════════════════════════════════════════════
   TRAIN SCENE — placeholder for the new practice mode
════════════════════════════════════════════════════════════════════════════ */
/* ── Difficulty → timing for Tréning ─────────────────────────────────────────
   Reuses the same 0–9 OBTIAŽNOSŤ knob shown at the bottom of every screen, so
   it means the same thing everywhere: 0 = most forgiving, 9 = hardest.
     time limit    14s (lvl 0)  →  6s  (lvl 9)
     hint interval 3.2s (lvl 0) →  1.7s (lvl 9)  — how often a new letter appears
*/
function trainTiming(level) {
  const t = Phaser.Math.Clamp(level, 0, 9) / 9;
  return {
    timeLimitMs: Math.round(14000 - t * 8000),
    hintMs:      Math.round(3200 - t * 1500),
  };
}

// Long words/phrases ("virtual reality headset") must not run off the 720px-
// wide screen. Shrink font size (and, for the letter-spaced mask, the gap
// between letters too) as the text gets longer, instead of a single fixed size.
function fitFontSize(len, base) {
  if (len > 26) return Math.round(base * 0.55);
  if (len > 20) return Math.round(base * 0.65);
  if (len > 14) return Math.round(base * 0.8);
  if (len > 9)  return Math.round(base * 0.92);
  return base;
}
function fitLetterSpacing(len) {
  if (len > 20) return 2;
  if (len > 14) return 4;
  if (len > 9)  return 5;
  return 7;
}

/* ════════════════════════════════════════════════════════════════════════════
   TRAIN SCENE — typing quiz with a letter-by-letter hint, no multiple choice.
   ---------------------------------------------------------------------------
   Deliberately NOT multiple-choice: picking from 4 options lets a child pass
   by elimination/pattern-matching without really knowing the word. Here they
   always have to type the real answer — the hint (one more letter revealed
   every couple of seconds) is just a safety net so nobody gets stuck staring
   at a blank line, not an answer key.
════════════════════════════════════════════════════════════════════════════ */
class TrainScene extends Phaser.Scene {
  constructor() { super('Train'); }

  init(data) {
    this.subject   = data.subject || 'english';
    this.ids       = data.ids || [];
    this.back      = data.back || { scene: 'Menu', data: {} };
    this._poolOverride = data._pool || null;   // set by retryMissed()
  }

  create() {
    this.cfg    = SUBJECTS[this.subject];
    this.pool   = this._poolOverride
      ? Phaser.Utils.Array.Shuffle(this._poolOverride.slice())
      : Phaser.Utils.Array.Shuffle(wordsForSelection(this.subject, this.ids).slice());
    this.words  = this.pool;
    this.total  = this.words.length;
    this.idx    = 0;
    this.score  = 0;
    this.missed = [];
    this.vkOn   = vkEnabled;

    menuHeader(this, '⚡  Tréning', selectionLabel(this.subject, this.ids), this.cfg.color,
      () => this.leave());

    if (!this.words.length) {
      this.add.text(W / 2, 260, 'Táto lekcia nemá žiadne slovíčka.', {
        fontSize: '16px', fill: '#445566', fontFamily: 'Arial, sans-serif',
      }).setOrigin(0.5);
      const b = makeButton(this, W / 2, 340, 260, 46, 0x1e2a3a, '← späť na lekcie', 18);
      b.bg.setStrokeStyle(2, 0x334455); b.txt.setStyle({ fill: '#9aaabb' });
      b.bg.on('pointerdown', () => this.leave());
      return;
    }

    this.progressTxt = this.add.text(W / 2, 82, '', {
      fontSize: '12px', fill: '#667788', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5, 0);

    // Time bar
    this.barBg = this.add.rectangle(W / 2, 106, 600, 8, 0x1c2a3a).setStrokeStyle(1, 0x2a3c50);
    this.bar   = this.add.rectangle(W / 2 - 300, 106, 600, 8, this.cfg.colorHex).setOrigin(0, 0.5);

    this.promptSideTxt = this.add.text(W / 2, 140, '', {
      fontSize: '12px', fill: '#667788', fontFamily: 'Arial, sans-serif', letterSpacing: 3,
    }).setOrigin(0.5, 0);
    this.promptTxt = this.add.text(W / 2, 168, '', {
      fontSize: '34px', fill: '#ffffff', fontFamily: '"Arial Black", Arial, sans-serif',
      align: 'center', wordWrap: { width: 680 },
    }).setOrigin(0.5, 0);

    this.maskTxt = this.add.text(W / 2, 240, '', {
      fontSize: '30px', fill: this.cfg.color, fontFamily: '"Courier New", monospace',
      align: 'center', wordWrap: { width: 680 },
    }).setOrigin(0.5, 0);

    this.typedTxt = this.add.text(W / 2, 288, '', {
      fontSize: '16px', fill: '#8899aa', fontFamily: '"Courier New", monospace',
      align: 'center', wordWrap: { width: 680 },
    }).setOrigin(0.5, 0);

    this.feedbackTxt = this.add.text(W / 2, 320, '', {
      fontSize: '15px', fill: '#66ff99', fontFamily: '"Arial Black", Arial, sans-serif',
    }).setOrigin(0.5, 0);

    // Physical + on-screen keyboard funnel through the same two functions.
    this.onKey = e => {
      if (this.locked) return;
      if (e.key === 'Backspace') { this.pressBackspace(); return; }
      if (e.key === 'Enter') return;
      if (!/^[a-zA-Zá-žÁ-Ž]$/.test(e.key)) return;
      this.pressLetter(e.key);
    };
    this.input.keyboard.on('keydown', this.onKey, this);
    this.events.once('shutdown', () => this.input.keyboard.off('keydown', this.onKey, this));

    if (this.vkOn) this.buildVirtualKeyboard(360);

    this.startQuestion();
  }

  leave() {
    if (this.hintTimer) this.hintTimer.remove(false);
    if (this.timeoutTimer) this.timeoutTimer.remove(false);
    this.scene.start(this.back.scene, this.back.data);
  }

  /* ── One question ── */
  startQuestion() {
    this.locked = false;
    const pair = this.words[this.idx];
    // Random direction each question, so both EN→SK and SK→EN get practised.
    this.toSlovak = Math.random() < 0.5;
    this.prompt   = this.toSlovak ? pair.en : pair.sk;
    this.target   = this.toSlovak ? pair.sk : pair.en;
    this.pair     = pair;
    this.typed    = '';
    this.revealed = 0;       // how many letters of the target are shown as hints
    this.skeleton = false;   // has the "_ _ _" outline itself appeared yet?

    const { timeLimitMs, hintMs } = trainTiming(MENU_STATE.level);
    this.timeLimitMs = timeLimitMs;

    this.progressTxt.setText(`Otázka ${this.idx + 1} / ${this.total}   ·   skóre ${this.score}`);
    this.promptSideTxt.setText(this.toSlovak ? 'PRELOŽ DO SLOVENČINY' : 'PRELOŽ DO ANGLIČTINY');
    this.promptTxt.setText(this.prompt)
      .setStyle({ fontSize: fitFontSize(this.prompt.length, 34) + 'px' });
    this.feedbackTxt.setText('');
    this.renderMask();
    this.renderTyped();

    this.bar.width = 600;
    this.bar.setFillStyle(this.cfg.colorHex);
    this.elapsed = 0;
    this.tickTimer = this.time.addEvent({
      delay: 50, loop: true, callback: () => {
        this.elapsed += 50;
        this.bar.width = Math.max(0, 600 * (1 - this.elapsed / this.timeLimitMs));
      },
    });
    /* First tick just reveals the "_ _ _" outline (how many letters the word
       has) — no actual letter yet. Every tick after that reveals one more
       letter. So: nothing for the first hintMs, then the skeleton, then
       letters — giving a real no-hint window for pure recall first. */
    this.hintTimer = this.time.addEvent({
      delay: hintMs, loop: true, callback: () => {
        if (!this.skeleton) { this.skeleton = true; this.renderMask(); }
        else this.revealNext();
      },
    });
    this.timeoutTimer = this.time.delayedCall(this.timeLimitMs, () => this.onTimeout());
  }

  nonSpaceLen(s) { return s.replace(/\s/g, '').length; }

  renderMask() {
    const size = fitFontSize(this.target.length, 30);
    this.maskTxt.setStyle({ fontSize: size + 'px', letterSpacing: fitLetterSpacing(this.target.length) });
    if (!this.skeleton) { this.maskTxt.setText(''); return; }
    let shown = 0;
    const out = [...this.target].map(ch => {
      if (ch === ' ') return ' ';
      shown += 1;
      return shown <= this.revealed ? ch : '_';
    }).join('');
    this.maskTxt.setText(out);
  }

  renderTyped() {
    this.typedTxt.setText(this.typed ? `Píšeš: ${this.typed}` : '');
  }

  revealNext() {
    const max = this.nonSpaceLen(this.target);
    if (this.revealed >= max) return;
    this.revealed += 1;
    this.renderMask();
  }

  // Same behaviour as the falling-word game: a keystroke is only accepted if
  // it keeps `typed` a valid prefix of the answer. A wrong key is rejected
  // (not typed) AND costs 1s off the clock — otherwise a child could just
  // mash every key and land on the right one by brute force instead of
  // actually knowing the word. Backspace is never penalised.
  pressLetter(letter) {
    if (this.locked) return;
    const cand = this.typed + letter.toLowerCase();
    if (!normalize(this.target).startsWith(normalize(cand))) {
      this.applyMistakePenalty();
      if (!this.locked) this.flashTypo();
      return;
    }
    this.typed = cand;
    this.renderTyped();
    if (normalize(this.typed) === normalize(this.target)) this.onCorrect();
  }

  applyMistakePenalty() {
    const PENALTY_MS = 1000;
    this.elapsed = Math.min(this.timeLimitMs, this.elapsed + PENALTY_MS);
    this.bar.width = Math.max(0, 600 * (1 - this.elapsed / this.timeLimitMs));
    this.penaltyPop();
    if (this.elapsed >= this.timeLimitMs) this.onTimeout();
  }

  penaltyPop() {
    const t = this.add.text(W / 2 + this.bar.width - 300, 96, '−1s', {
      fontSize: '13px', fill: '#ff6666', fontFamily: '"Arial Black", Arial, sans-serif',
    }).setOrigin(0.5);
    this.tweens.add({
      targets: t, y: 78, alpha: { from: 1, to: 0 }, duration: 500,
      onComplete: () => t.destroy(),
    });
  }

  pressBackspace() {
    if (this.locked || !this.typed) return;
    this.typed = this.typed.slice(0, -1);
    this.renderTyped();
  }

  flashTypo() {
    this.typedTxt.setStyle({ fill: '#ff6666' });
    this.tweens.add({
      targets: this.typedTxt, x: { from: W / 2 - 8, to: W / 2 + 8 }, duration: 40, yoyo: true, repeat: 3,
      onComplete: () => { this.typedTxt.setStyle({ fill: '#8899aa' }); this.typedTxt.x = W / 2; },
    });
  }

  clearTimers() {
    if (this.tickTimer)    this.tickTimer.remove(false);
    if (this.hintTimer)    this.hintTimer.remove(false);
    if (this.timeoutTimer) this.timeoutTimer.remove(false);
  }

  // Shows the answer, spelled out, in the same font/spacing rules as the mask.
  revealFullTarget() {
    this.skeleton = true;
    this.revealed = this.nonSpaceLen(this.target);
    this.renderMask();
  }

  onCorrect() {
    this.locked = true;
    this.clearTimers();
    this.score += 1;
    this.feedbackTxt.setText('✓ správne!').setStyle({ fill: '#66ff99' });
    this.revealFullTarget();
    this.time.delayedCall(650, () => this.advance());

  }

  onTimeout() {
    this.locked = true;
    this.clearTimers();
    this.missed.push(this.pair);
    this.bar.width = 0;
    this.feedbackTxt.setText(`✗ správne bolo: ${this.target}`).setStyle({ fill: '#ff8866' });
    this.revealFullTarget();
    this.time.delayedCall(1400, () => this.advance());
  }

  advance() {
    this.idx += 1;
    if (this.idx >= this.words.length) this.showResults();
    else this.startQuestion();
  }

  showResults() {
    [this.progressTxt, this.barBg, this.bar, this.promptSideTxt, this.promptTxt,
     this.maskTxt, this.typedTxt, this.feedbackTxt].forEach(o => o && o.setVisible(false));
    if (this.vkKeys) this.vkKeys.forEach(k => k.destroy());

    const n = this.words.length;
    this.add.text(W / 2, 130, `${this.score} / ${n}`, {
      fontSize: '54px', fill: this.cfg.color, fontFamily: '"Arial Black", Arial, sans-serif',
    }).setOrigin(0.5);
    this.add.text(W / 2, 190, this.score === n ? 'Všetko správne! 🎉' : 'Hotovo', {
      fontSize: '16px', fill: '#aabbcc', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5);

    if (this.missed.length) {
      this.add.text(W / 2, 230, 'Pomýlil(a) si sa v:', {
        fontSize: '12px', fill: '#8899aa', fontFamily: 'Arial, sans-serif',
      }).setOrigin(0.5);
      this.missed.slice(0, 8).forEach((p, i) => {
        this.add.text(W / 2, 256 + i * 22, `${p.en}  →  ${p.sk}`, {
          fontSize: '14px', fill: '#dde6ee', fontFamily: 'Arial, sans-serif',
        }).setOrigin(0.5);
      });
      if (this.missed.length > 8) {
        this.add.text(W / 2, 256 + 8 * 22, `… a ${this.missed.length - 8} ďalších`, {
          fontSize: '12px', fill: '#556677', fontFamily: 'Arial, sans-serif',
        }).setOrigin(0.5);
      }
    }

    const btnY = 470;
    if (this.missed.length) {
      const rb = makeButton(this, W / 2 - 150, btnY, 280, 46, 0x0066bb, 'Skúsiť znova chybné', 17);
      rb.bg.on('pointerdown', () => this.retryMissed());
      const bb = makeButton(this, W / 2 + 150, btnY, 220, 46, 0x1e2a3a, '← späť na lekcie', 17);
      bb.bg.setStrokeStyle(2, 0x334455); bb.txt.setStyle({ fill: '#9aaabb' });
      bb.bg.on('pointerdown', () => this.leave());
    } else {
      const bb = makeButton(this, W / 2, btnY, 260, 46, 0x1e2a3a, '← späť na lekcie', 18);
      bb.bg.setStrokeStyle(2, 0x334455); bb.txt.setStyle({ fill: '#9aaabb' });
      bb.bg.on('pointerdown', () => this.leave());
    }
  }

  retryMissed() {
    this.scene.restart({ subject: this.subject, ids: this.ids, back: this.back, _pool: this.missed });
  }

  // On-screen QWERTY — same layout/feel as the falling-word game's keyboard,
  // funnelled through pressLetter()/pressBackspace() like everywhere else.
  buildVirtualKeyboard(topY) {
    const GAP = 6, KEY_H = 46;
    const KEY_W = Math.floor((W - 32 - 9 * GAP) / 10);
    const rows = [
      [...'qwertyuiop'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => this.pressLetter(ch) })),
      [...'asdfghjkl'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => this.pressLetter(ch) })),
      [...'zxcvbnm'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => this.pressLetter(ch) }))
        .concat([{ label: '⌫', w: KEY_W * 1.7, fn: () => this.pressBackspace(), fill: 0x3a1c1c }]),
    ];
    this.vkKeys = [];
    const makeKey = (x, y, spec) => {
      const fill = spec.fill || 0x1c2a3a;
      const bg = this.add.rectangle(x, y, spec.w, KEY_H, fill)
        .setStrokeStyle(1, 0x334455).setInteractive({ useHandCursor: true });
      const txt = this.add.text(x, y, spec.label, {
        fontSize: '18px', fill: '#dde8ff', fontFamily: '"Arial Black", Arial, sans-serif',
      }).setOrigin(0.5);
      bg.on('pointerdown', () => { bg.setFillStyle(0x33475e); spec.fn(); });
      bg.on('pointerup',   () => bg.setFillStyle(fill));
      bg.on('pointerout',  () => bg.setFillStyle(fill));
      this.vkKeys.push(bg, txt);
    };
    rows.forEach((row, r) => {
      const y = topY + r * (KEY_H + GAP);
      const rowWidth = row.reduce((s, k) => s + k.w, 0) + (row.length - 1) * GAP;
      let x = (W - rowWidth) / 2 + row[0].w / 2;
      row.forEach((spec, i) => {
        makeKey(x, y, spec);
        if (i < row.length - 1) x += spec.w / 2 + GAP + row[i + 1].w / 2;
      });
    });
  }
}

/* ════════════════════════════════════════════════════════════════════════════
   GAME SCENE
════════════════════════════════════════════════════════════════════════════ */
class GameScene extends Phaser.Scene {
  constructor() { super('Game'); }

  init(data) {
    this.startLevel  = typeof data.level === 'number' ? data.level : 3;
    this.kidMode     = data.kid  || false;
    this.gameMode    = data.mode || 'typing'; // mechanic: 'typing' | 'translation' | 'story'
    this.unit        = data.unit || null;      // lesson id (a key in trickyUnits), or null for Random Words
    // Which leaderboard category this run belongs to, and which menu screen
    // the ⌂ / game-over buttons should return to. Both come from the menu so
    // the game engine doesn't need to know about subjects at all.
    this.lbMode      = data.lbMode || lbModeFor(this.gameMode, this.unit);
    this.backTo      = data.back || { scene: 'Menu', data: {} };
  }

  /* ── helpers ── */
  getEffectiveCfg(level) {
    const b = LEVEL_CONFIG[level];
    let cfg;
    if (this.gameMode === 'translation' || this.gameMode === 'story') {
      const kidMul = this.kidMode ? 2 : 1;
      cfg = { speed: b.speed * 0.7 / kidMul, interval: b.interval * 2 * kidMul, pool: b.pool };
    } else {
      cfg = this.kidMode
        ? { speed: b.speed * 0.5, interval: b.interval * 2, pool: b.pool }
        : b;
    }
    // The on-screen keyboard eats vertical space, shortening the fall
    // distance. Scale speed down proportionally so words take about the
    // same TIME to land as on desktop, instead of the shrunken screen
    // silently making the game harder.
    return this.vkOn ? { ...cfg, speed: cfg.speed * (this.groundY / GROUND_Y) } : cfg;
  }

  buildPool() {
    if (this.gameMode === 'translation' || this.gameMode === 'story') {
      let source;
      if (this.gameMode === 'story') {
        // No pictures loaded yet for this unit? Fall back to its plain word
        // list so the mode is still testable — spawnStoryWord() shows a 🖼️
        // placeholder instead of a real image for entries with no imageKey.
        const withPics = storyUnits[this.unit] || [];
        source = withPics.length ? withPics : (trickyUnits[this.unit] || []);
      } else {
        // Translation mechanic: a specific Tricky Words unit, or the
        // generic Random Words vocabulary when no unit is set.
        source = this.unit ? (trickyUnits[this.unit] || []) : typingTranslationPool;
      }
      this.transPool = Phaser.Utils.Array.Shuffle(source.length ? [...source] : [{ en: 'word', sk: 'slovo' }]);
      this.transIdx  = 0;
      return;
    }
    const cfg      = this.getEffectiveCfg(this.currentLevel);
    const filtered = wordPool.filter(w => cfg.pool.includes(w.level)).map(w => w.word);
    this.pool      = Phaser.Utils.Array.Shuffle(filtered.length ? filtered : ['type','word','game','play','test']);
    this.poolIdx   = 0;
  }

  nextWord() {
    const w = this.pool[this.poolIdx % this.pool.length];
    this.poolIdx++;
    if (this.poolIdx >= this.pool.length) { Phaser.Utils.Array.Shuffle(this.pool); this.poolIdx = 0; }
    return w;
  }

  /* ── create ── */
  create() {
    this.vkOn    = vkEnabled;
    this.groundY = this.vkOn ? 360 : GROUND_Y; // shrink play area to make room for the on-screen keyboard

    this.hp           = MAX_HP;
    this.score        = 0;
    this.typed        = '';
    this.words        = [];
    this.flashing     = false;
    this.dead         = false;
    this.frozen       = true;   // true during countdown / pause
    this.gamePaused   = false;
    this.pauseElems   = [];
    this.currentLevel = this.startLevel;

    const cfg = this.getEffectiveCfg(this.currentLevel);
    this.fallSpeed = cfg.speed;

    /* Background */
    this.bgRect = this.add.rectangle(W / 2, H / 2, W, H, 0x1a1a2e).setDepth(0);
    drawGrid(this, 0x1e2640, 0.4);

    /* Ground line */
    this.add.rectangle(W / 2, this.groundY + 2, W, 4, 0xff3333).setDepth(1);

    /* ── HP bar (top-left) ── */
    this.hpGfx = this.add.graphics().setDepth(3);
    this.HP    = { x: 14, y: 13, w: 180, h: 22 };
    this.drawHpBar();
    this.hpLbl = this.add.text(14, this.HP.y + this.HP.h + 2, `HP: ${this.hp}`, {
      fontSize: '14px', fill: '#cc4444', fontFamily: 'Arial, sans-serif',
    }).setDepth(3);

    /* ── Level display (top-center) ── */
    this.levelTxt = this.add.text(W / 2, 12, '', {
      fontSize: '22px', fill: LEVEL_COLORS[this.currentLevel],
      fontFamily: '"Arial Black", Arial, sans-serif',
      stroke: '#000033', strokeThickness: 3,
    }).setOrigin(0.5, 0).setDepth(3);

    this.lvlUpTxt = this.add.text(W / 2, 38, '', {
      fontSize: '12px', fill: '#445566', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5, 0).setDepth(3);

    this.refreshLevelDisplay();

    /* ── Score (top-right) ── */
    this.scoreTxt = this.add.text(W - 14, 13, 'Score: 0', {
      fontSize: '22px', fill: '#ffdd00',
      fontFamily: '"Arial Black", Arial, sans-serif',
      stroke: '#443300', strokeThickness: 3,
    }).setOrigin(1, 0).setDepth(3);

    /* ── Menu button (top-right, below score) ── */
    this.menuConfirm = false;
    const menuBtnBg = this.add.rectangle(W - 57, 46, 88, 22, 0x1c2a3a)
      .setStrokeStyle(1, 0x334455).setInteractive({ useHandCursor: true }).setDepth(3);
    this.menuBtnTxt = this.add.text(W - 57, 46, '⌂ MENU', {
      fontSize: '12px', fill: '#7788aa', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5).setDepth(3);
    menuBtnBg.on('pointerover', () => { menuBtnBg.setFillStyle(0x263444); if (!this.menuConfirm) this.menuBtnTxt.setStyle({ fill: '#aabbcc' }); });
    menuBtnBg.on('pointerout',  () => { menuBtnBg.setFillStyle(0x1c2a3a); if (!this.menuConfirm) this.menuBtnTxt.setStyle({ fill: '#7788aa' }); });
    menuBtnBg.on('pointerdown', () => {
      if (!this.menuConfirm) {
        this.menuConfirm = true;
        this.menuBtnTxt.setText('CONFIRM?').setStyle({ fill: '#ff8844' });
        // auto-reset after 2s if not clicked again
        this.time.delayedCall(2000, () => {
          if (this.menuConfirm) { this.menuConfirm = false; this.menuBtnTxt.setText('⌂ MENU').setStyle({ fill: '#7788aa' }); }
        });
      } else {
        this.dead = true;
        this.spawnTimer.remove(false);
        if (this.levelTimer) this.levelTimer.remove(false);
        this.input.keyboard.off('keydown', this.onKey, this);
        for (const fw of this.words) fw.obj.destroy();
        this.words = [];
        this.scene.start(this.backTo.scene, this.backTo.data);
      }
    });

    /* ── Pause button (top-right, left of menu button) ── */
    const pauseBtnBg = this.add.rectangle(W - 155, 46, 80, 22, 0x1c2a3a)
      .setStrokeStyle(1, 0x334455).setInteractive({ useHandCursor: true }).setDepth(3);
    this.pauseBtnTxt = this.add.text(W - 155, 46, '⏸', {
      fontSize: '14px', fill: '#7788aa', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5).setDepth(3);
    pauseBtnBg.on('pointerover', () => pauseBtnBg.setFillStyle(0x263444));
    pauseBtnBg.on('pointerout',  () => pauseBtnBg.setFillStyle(0x1c2a3a));
    pauseBtnBg.on('pointerdown', () => {
      if (this.gamePaused) this.resumeGame();
      else this.pauseGame();
    });

    /* ── Input bar (bottom) + optional on-screen keyboard ── */
    const barTop = this.groundY + 36;
    this.add.rectangle(W / 2, (barTop + H) / 2, W, H - barTop, 0x0d1020).setDepth(1);
    this.add.rectangle(W / 2, barTop, W, 2, 0x1e2e50).setDepth(2);

    const inputY = this.vkOn ? barTop + 20 : H - 27;
    this.inputTxt = this.add.text(W / 2, inputY, '', {
      fontSize: this.vkOn ? '22px' : '28px', fill: '#00ff88', fontFamily: 'monospace',
      stroke: '#003322', strokeThickness: 2,
    }).setOrigin(0.5).setDepth(3);

    if (this.vkOn) this.buildVirtualKeyboard(barTop + 34);

    /* Keyboard */
    this.input.keyboard.on('keydown', this.onKey, this);

    /* Countdown 3-2-1-GO!, then begin play */
    this.startCountdown(() => this.beginPlay());
  }

  // On-screen QWERTY keyboard (no diacritics needed — normalize() already
  // accepts e.g. "macka" for "mačka"). Both this and the physical keyboard
  // funnel through pressLetter()/pressBackspace() so the matching/scoring
  // logic only lives in one place.
  buildVirtualKeyboard(topY) {
    const GAP = 6, KEY_H = 46;
    const KEY_W = Math.floor((W - 32 - 9 * GAP) / 10); // sized off the 10-key top row

    const rows = [
      [...'qwertyuiop'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => this.pressLetter(ch) })),
      [...'asdfghjkl'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => this.pressLetter(ch) })),
      [...'zxcvbnm'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => this.pressLetter(ch) }))
        .concat([{ label: '⌫', w: KEY_W * 1.7, fn: () => this.pressBackspace(), fill: 0x3a1c1c }]),
    ];

    const makeKey = (x, y, spec) => {
      const fill = spec.fill || 0x1c2a3a;
      const bg = this.add.rectangle(x, y, spec.w, KEY_H, fill)
        .setStrokeStyle(1, 0x334455).setDepth(6).setInteractive({ useHandCursor: true });
      this.add.text(x, y, spec.label, {
        fontSize: '18px', fill: '#dde8ff', fontFamily: '"Arial Black", Arial, sans-serif',
      }).setOrigin(0.5).setDepth(7);
      bg.on('pointerdown', () => {
        if (this.dead || this.frozen) return;
        bg.setFillStyle(0x33475e); spec.fn();
      });
      bg.on('pointerup',   () => bg.setFillStyle(fill));
      bg.on('pointerout',  () => bg.setFillStyle(fill));
    };

    rows.forEach((row, r) => {
      const y = topY + r * (KEY_H + GAP);
      const rowWidth = row.reduce((s, k) => s + k.w, 0) + (row.length - 1) * GAP;
      let x = (W - rowWidth) / 2 + row[0].w / 2;
      row.forEach((spec, i) => {
        makeKey(x, y, spec);
        if (i < row.length - 1) x += spec.w / 2 + GAP + row[i + 1].w / 2;
      });
    });
  }

  restartSpawnTimer(interval) {
    if (this.spawnTimer) this.spawnTimer.remove(false);
    this.spawnTimer = this.time.addEvent({
      delay: interval, callback: this.spawnWord, callbackScope: this, loop: true,
    });
  }

  beginPlay() {
    this.frozen = false;
    const cfg = this.getEffectiveCfg(this.currentLevel);
    this.buildPool();
    this.restartSpawnTimer(cfg.interval);
    this.spawnWord();
    this.levelTimer = (this.currentLevel < 9)
      ? this.time.addEvent({ delay: LEVEL_UP_SEC * 1000, callback: this.levelUp, callbackScope: this, loop: true })
      : null;
  }

  startCountdown(onComplete) {
    const overlay = this.add.rectangle(W / 2, H / 2, W, H, 0x000000).setAlpha(0.5).setDepth(18);
    const STEP   = 850;
    const labels = ['3', '2', '1', 'GO!'];
    const colors = ['#ff6644', '#ffcc00', '#88ff44', '#00ff88'];
    labels.forEach((label, i) => {
      this.time.delayedCall(i * STEP, () => {
        const isGo = i === 3;
        const t = this.add.text(W / 2, H / 2, label, {
          fontSize: isGo ? '78px' : '120px',
          fill: colors[i],
          fontFamily: '"Arial Black", Arial, sans-serif',
          stroke: '#000033', strokeThickness: 10,
          shadow: { color: colors[i], blur: 40, fill: true },
        }).setOrigin(0.5).setDepth(20);
        this.tweens.add({
          targets: t,
          alpha: { from: 1, to: 0 },
          scale: { from: 1, to: isGo ? 1.8 : 0.6 },
          duration: STEP - 80,
          ease: 'Power2',
          onComplete: () => t.destroy(),
        });
      });
    });
    this.time.delayedCall(labels.length * STEP, () => { overlay.destroy(); if (onComplete) onComplete(); });
  }

  pauseGame() {
    if (this.dead || this.frozen) return;
    this.frozen = true;
    this.gamePaused = true;
    this.pauseBtnTxt.setText('▶ RESUME').setStyle({ fill: '#aaffaa' });
    this.pauseElems.push(
      this.add.rectangle(W / 2, H / 2, W, H, 0x000000).setAlpha(0.55).setDepth(15),
      this.add.text(W / 2, H / 2 - 20, 'PAUSED', {
        fontSize: '72px', fill: '#ffffff',
        fontFamily: '"Arial Black", Arial, sans-serif',
        stroke: '#000033', strokeThickness: 8,
      }).setOrigin(0.5).setDepth(16),
      this.add.text(W / 2, H / 2 + 46, 'click  ▶ RESUME  to continue', {
        fontSize: '18px', fill: '#8899aa', fontFamily: 'Arial, sans-serif',
      }).setOrigin(0.5).setDepth(16),
    );
  }

  resumeGame() {
    this.gamePaused = false;
    this.pauseBtnTxt.setText('⏸').setStyle({ fill: '#7788aa' });
    for (const e of this.pauseElems) e.destroy();
    this.pauseElems = [];
    this.startCountdown(() => { this.frozen = false; });
  }

  /* ── Spawn ── */
  spawnWord() {
    if (this.dead || this.frozen) return;
    if (this.gameMode === 'story') { this.spawnStoryWord(); return; }
    if (this.gameMode === 'translation') { this.spawnTranslationWord(); return; }

    const active = new Set(this.words.map(w => w.word));
    let word, tries = 0;
    do { word = this.nextWord(); tries++; } while (active.has(word) && tries < 20);

    const obj = this.add.text(
      Phaser.Math.Between(70, W - 70), -34, word, {
        fontSize: '30px', fill: '#e0e8ff', fontFamily: 'monospace',
        stroke: '#000022', strokeThickness: 4,
      }
    ).setOrigin(0.5, 0).setDepth(4);
    obj.x = Phaser.Math.Clamp(obj.x, obj.width / 2 + 8, W - obj.width / 2 - 8);
    this.words.push({ obj, word, missedLabel: capWords(word) });
  }

  // Story mode: a picture falls instead of a text prompt. The answer is always
  // the English word (no random EN/SK direction like Tricky/Translation).
  spawnStoryWord() {
    const activeEN = new Set(this.words.map(w => w.pairEN));
    const pool = this.transPool;
    let pair, tries = 0;
    do {
      pair = pool[this.transIdx % pool.length];
      this.transIdx++;
      if (this.transIdx >= pool.length) { Phaser.Utils.Array.Shuffle(pool); this.transIdx = 0; }
      tries++;
    } while (activeEN.has(pair.en) && tries < 20);

    const answer = pair.en;
    const IMG = 110;
    const x = Phaser.Math.Between(100, W - 100);
    const container = this.add.container(x, -(IMG + 40)).setDepth(4);

    const frame = this.add.rectangle(0, IMG / 2, IMG + 6, IMG + 6, 0x000000, 0)
      .setStrokeStyle(2, 0x334466);

    const pic = [];
    if (pair.imageKey) {
      pic.push(this.add.image(0, IMG / 2, pair.imageKey).setDisplaySize(IMG, IMG));
    } else {
      // No picture for this word yet — placeholder so the mode stays fully
      // playable while pictures are still being added to images/story/.
      pic.push(this.add.rectangle(0, IMG / 2, IMG, IMG, 0x1c2a3a).setStrokeStyle(1, 0x2a3a4a));
      pic.push(this.add.text(0, IMG / 2, '🖼️', { fontSize: '40px' }).setOrigin(0.5));
    }

    // one underscore per letter, triple-space between words
    const underStr = answer.split(' ').map(w => [...w].map(() => '_').join(' ')).join('   ');
    const underTxt = this.add.text(0, IMG + 16, underStr, {
      fontSize: '15px', fill: '#334466', fontFamily: 'monospace',
    }).setOrigin(0.5, 0);

    container.add([...pic, frame, underTxt]);
    const maxHalf = Math.max(IMG / 2 + 3, underTxt.width / 2);
    container.x = Phaser.Math.Clamp(container.x, maxHalf + 10, W - maxHalf - 10);
    // isTranslation reused here so the shared highlight/reset/damage logic
    // (which only knows about "container + underTxt" words) works unchanged.
    this.words.push({
      obj: container, word: answer, pairEN: pair.en, underTxt, isTranslation: true,
      missedLabel: `${capWords(pair.en)} → ${capWords(pair.sk)}`,
    });
  }

  spawnTranslationWord() {
    const activeEN = new Set(this.words.map(w => w.pairEN));
    const pool = this.transPool;
    let pair, tries = 0;
    do {
      pair = pool[this.transIdx % pool.length];
      this.transIdx++;
      if (this.transIdx >= pool.length) { Phaser.Utils.Array.Shuffle(pool); this.transIdx = 0; }
      tries++;
    } while (activeEN.has(pair.en) && tries < 20);

    const enToSk  = Math.random() < 0.5;
    const prompt  = enToSk ? pair.en : pair.sk;
    const answer  = enToSk ? pair.sk : pair.en;
    const sides   = unitSideLabels(this.unit);
    const label   = enToSk ? `${sides.left} → ${sides.right}` : `${sides.right} → ${sides.left}`;

    const x = Phaser.Math.Between(100, W - 100);
    const container = this.add.container(x, -80).setDepth(4);

    const lblTxt = this.add.text(0, 0, label, {
      fontSize: '11px', fill: '#445577', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5, 0);

    const promptTxt = this.add.text(0, 14, prompt, {
      fontSize: '24px', fill: '#aabbff', fontFamily: 'monospace',
      stroke: '#000022', strokeThickness: 3,
    }).setOrigin(0.5, 0);

    // one underscore per letter, triple-space between words
    const underStr = answer.split(' ').map(w => [...w].map(() => '_').join(' ')).join('   ');
    const underTxt = this.add.text(0, 44, underStr, {
      fontSize: '15px', fill: '#334466', fontFamily: 'monospace',
    }).setOrigin(0.5, 0);

    container.add([lblTxt, promptTxt, underTxt]);
    // Clamp so the widest child never overflows the screen
    const maxHalf = Math.max(lblTxt.width, promptTxt.width, underTxt.width) / 2;
    container.x = Phaser.Math.Clamp(container.x, maxHalf + 10, W - maxHalf - 10);
    this.words.push({
      obj: container, word: answer, pairEN: pair.en, underTxt, promptTxt, isTranslation: true,
      missedLabel: `${capWords(pair.en)} → ${capWords(pair.sk)}`,
    });
  }

  /* ── Level up ── */
  levelUp() {
    if (this.dead || this.frozen || this.currentLevel >= 9) return;
    this.currentLevel = Math.min(9, this.currentLevel + 1);
    const cfg = this.getEffectiveCfg(this.currentLevel);
    this.fallSpeed = cfg.speed;
    this.restartSpawnTimer(cfg.interval);
    this.buildPool();
    this.refreshLevelDisplay();
    this.showLevelUpNotif();
    if (this.currentLevel >= 9 && this.levelTimer) { this.levelTimer.remove(false); this.levelTimer = null; }
  }

  refreshLevelDisplay() {
    this.levelTxt.setText(`LVL ${this.currentLevel}`).setStyle({ fill: LEVEL_COLORS[this.currentLevel] });
  }

  showLevelUpNotif() {
    const n = this.add.text(W / 2, H / 2 - 60, `⬆ LEVEL ${this.currentLevel}`, {
      fontSize: '50px', fill: LEVEL_COLORS[this.currentLevel],
      fontFamily: '"Arial Black", Arial, sans-serif',
      stroke: '#000033', strokeThickness: 6,
    }).setOrigin(0.5).setDepth(10);
    this.tweens.add({ targets: n, y: n.y - 90, alpha: 0, duration: 1400, ease: 'Cubic.Out', onComplete: () => n.destroy() });
  }

  /* ── Key handler ── */
  onKey(e) {
    if (this.dead || this.frozen) return;
    if (e.key === 'Backspace') { this.pressBackspace(); return; }
    if (e.key === 'Escape')    { this.typed = ''; this.refreshInput(); this.highlight(); return; }
    // Allow basic Latin letters + diacritics (for Slovak in translation mode)
    if (e.key.length !== 1 || !/^\p{L}$/u.test(e.key)) return;
    this.pressLetter(e.key);
  }

  // Shared by the physical keyboard (onKey) and the on-screen virtual
  // keyboard (buildVirtualKeyboard) — the matching/scoring logic only
  // lives here, so both input methods behave identically.
  pressLetter(letter) {
    if (this.dead || this.frozen) return;
    const cand = this.typed + letter.toLowerCase();
    if (!this.words.some(w => normalize(w.word).startsWith(normalize(cand)))) { this.typed = cand; this.flashTypo(); }
    else { this.typed = cand; }

    const exact = this.words.find(w => normalize(w.word) === normalize(this.typed));
    if (exact) { this.destroyWord(exact); this.typed = ''; }

    this.refreshInput();
    this.highlight();
  }

  pressBackspace() {
    if (this.dead || this.frozen) return;
    this.typed = this.typed.slice(0, -1);
    this.refreshInput();
    this.highlight();
  }

  destroyWord(fw) {
    const pts = fw.word.length * 10 * (this.kidMode ? 2 : 1);
    this.score += pts;
    this.scoreTxt.setText(`Score: ${this.score}`);

    const popY = fw.isTranslation ? fw.obj.y + 40 : fw.obj.y + fw.obj.height / 2;
    const pop = this.add.text(fw.obj.x, popY, `+${pts}`, {
      fontSize: '22px', fill: '#ffff44',
      fontFamily: '"Arial Black", Arial, sans-serif',
      stroke: '#554400', strokeThickness: 3,
    }).setOrigin(0.5).setDepth(6);
    this.tweens.add({ targets: pop, y: pop.y - 70, alpha: 0, duration: 800, ease: 'Cubic.Out', onComplete: () => pop.destroy() });
    playSuccess();

    for (let i = 0; i < 6; i++) {
      const sp = this.add.circle(
        fw.obj.x + Phaser.Math.Between(-20, 20),
        popY,
        Phaser.Math.Between(3, 7),
        Phaser.Display.Color.HSLToColor(Math.random(), 1, 0.6).color,
      ).setDepth(5);
      this.tweens.add({ targets: sp, x: sp.x + Phaser.Math.Between(-60, 60), y: sp.y + Phaser.Math.Between(-60, 20), alpha: 0, scaleX: 0, scaleY: 0, duration: Phaser.Math.Between(400, 700), onComplete: () => sp.destroy() });
    }

    fw.obj.destroy();
    this.words = this.words.filter(w => w !== fw);
  }

  flashTypo() {
    if (this.flashing) return;
    this.flashing = true;
    this.bgRect.setFillStyle(0x550000);
    this.time.delayedCall(100, () => { this.bgRect.setFillStyle(0x1a1a2e); this.flashing = false; });
    playError();
  }

  refreshInput() { this.inputTxt.setText(this.typed ? `> ${this.typed}_` : ''); }

  highlight() {
    // Build underscore display — filled positions show the word's own chars (with diacritics)
    const buildUnder = (word, typed) => {
      const normTypedLen = normalize(typed).length;
      let charIdx = 0;
      return word.split(' ').map(part =>
        [...part].map(c => { const ch = charIdx < normTypedLen ? c : '_'; charIdx++; return ch; }).join(' ')
      ).join('   ');
    };

    const reset = fw => {
      if (fw.isTranslation) { fw.underTxt.setText(buildUnder(fw.word, '')).setStyle({ fill: '#334466' }); }
      else { fw.obj.setStyle({ fill: '#e0e8ff' }); }
    };
    const setCol = (fw, col) => fw.isTranslation ? fw.underTxt.setStyle({ fill: col }) : fw.obj.setStyle({ fill: col });

    for (const fw of this.words) reset(fw);
    if (!this.typed) return;
    const m = this.words.filter(w => normalize(w.word).startsWith(normalize(this.typed)));
    if (!m.length) { for (const fw of this.words) setCol(fw, '#884444'); return; }
    m.sort((a, b) => b.obj.y - a.obj.y);
    // Primary match: fill in typed characters
    setCol(m[0], '#00ff88');
    if (m[0].isTranslation) m[0].underTxt.setText(buildUnder(m[0].word, this.typed));
    for (let i = 1; i < m.length; i++) setCol(m[i], '#66ccaa');
  }

  drawHpBar() {
    const { x, y, w, h } = this.HP;
    const pct = Math.max(0, this.hp / MAX_HP);
    const col = pct > 0.5 ? 0x00cc55 : pct > 0.25 ? 0xdd7700 : 0xcc2222;
    this.hpGfx.clear();
    this.hpGfx.fillStyle(0x112233); this.hpGfx.fillRect(x, y, w, h);
    this.hpGfx.fillStyle(col);      this.hpGfx.fillRect(x, y, Math.ceil(w * pct), h);
    this.hpGfx.lineStyle(2, 0x223344); this.hpGfx.strokeRect(x, y, w, h);
  }

  updateHp() {
    this.drawHpBar();
    this.hpLbl.setText(`HP: ${this.hp}`);
    const pct = this.hp / MAX_HP;
    this.hpLbl.setStyle({ fill: pct > 0.5 ? '#55cc77' : pct > 0.25 ? '#dd8800' : '#ff3333' });
  }

  /* ── Update loop ── */
  update(_, delta) {
    if (this.dead || this.frozen) return;

    // Countdown display
    if (this.levelTimer && this.currentLevel < 9) {
      const sec = Math.ceil(this.levelTimer.getRemaining() / 1000);
      this.lvlUpTxt.setText(`→ LVL ${this.currentLevel + 1} in ${sec}s`);
    } else {
      this.lvlUpTxt.setText(this.currentLevel >= 9 ? '— MAX LEVEL —' : '');
    }

    const dt = delta / 1000, lost = [];
    for (const fw of this.words) {
      fw.obj.y += this.fallSpeed * dt;
      if (fw.obj.y >= this.groundY) lost.push(fw);
    }

    if (lost.length) {
      for (const fw of lost) {
        this.cameras.main.shake(180, 0.008);
        const normW   = normalize(fw.word);
        const normT   = normalize(this.typed);
        const already = normW.startsWith(normT) ? normT.length : 0;
        const dmg     = Math.max(1, normW.length - already);
        this.hp = Math.max(0, this.hp - dmg);
        const fl = this.add.rectangle(fw.obj.x, this.groundY + 2, 200, 6, 0xff6622).setDepth(5);
        this.time.delayedCall(240, () => fl.destroy());

        // Briefly recap the missed word/translation just above the ground
        // line so the player can read (and learn) it before it disappears.
        const recap = this.add.text(fw.obj.x, this.groundY - 10, fw.missedLabel || fw.word, {
          fontSize: '15px', fill: '#ff9966', fontFamily: 'Arial, sans-serif',
          stroke: '#331100', strokeThickness: 3,
        }).setOrigin(0.5, 1).setDepth(6);
        this.tweens.add({
          targets: recap, alpha: 0, y: recap.y - 10,
          delay: 1000, duration: 400, ease: 'Cubic.Out',
          onComplete: () => recap.destroy(),
        });

        fw.obj.destroy();
        this.words = this.words.filter(w => w !== fw);
      }
      // After removing whatever just fell, check the SURVIVING words only.
      // If the in-progress buffer no longer matches any of them, it's dead
      // weight — clear it so the very next keystroke can start a fresh word
      // immediately. If it still matches a surviving word, keep it (e.g.
      // "but" typed toward "button", which falls, but "butter" is still up).
      if (this.typed && !this.words.some(w => normalize(w.word).startsWith(normalize(this.typed)))) {
        this.typed = '';
        this.refreshInput();
      }
      this.updateHp();
      this.highlight();
      if (this.hp <= 0) { this.triggerGameOver(); }
    }
  }

  triggerGameOver() {
    this.dead = true;
    this.spawnTimer.remove(false);
    if (this.levelTimer) this.levelTimer.remove(false);
    this.input.keyboard.off('keydown', this.onKey, this);
    for (const fw of this.words) fw.obj.destroy();
    this.words = [];
    this.cameras.main.shake(400, 0.02);
    this.time.delayedCall(500, () => {
      this.scene.start('GameOver', {
        score:        this.score,
        startLevel:   this.startLevel,
        reachedLevel: this.currentLevel,
        kid:          this.kidMode,
        mechanic:     this.gameMode,
        unit:         this.unit,
        mode:         this.lbMode,
        lbMode:       this.lbMode,
        back:         this.backTo,
      });
    });
  }
}

class GameOverScene extends Phaser.Scene {

  /* ── Name entry ── */
  buildNameEntry() {
    const push = obj => { this.nameElems.push(obj); return obj; };

    push(this.add.text(W / 2, 196, '🏆  NEW HIGH SCORE!  You made Top 10!', {
      fontSize: '19px', fill: '#ffdd44',
      fontFamily: '"Arial Black", Arial, sans-serif',

    push(this.add.text(W / 2, 224, 'Enter your name:', {
      fontSize: '15px', fill: '#9aaabb', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5));

    push(this.add.rectangle(W / 2, 257, 360, 44, 0x0f1d30).setStrokeStyle(2, 0x0077cc));

    this.nameDisplay = push(this.add.text(W / 2, 257, '_', {
      fontSize: '26px', fill: '#ffffff', fontFamily: 'monospace',
    }).setOrigin(0.5));

    push(this.add.text(W / 2, 283, vkEnabled ? 'Tap ✓ OK to confirm  (max 16 characters)' : 'Press ENTER to confirm  (max 16 characters)', {
      fontSize: '12px', fill: '#3a4a5a', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5));

    this.cursorEv = this.time.addEvent({
      delay: 500, loop: true,
      callback: () => { this.cursorVis = !this.cursorVis; this.refreshNameDisplay(); },

  // On-screen keyboard for entering a leaderboard name on touch devices —
  // same visual style as GameScene's virtual keyboard, plus SPACE/✓ OK.
  // Elements are pushed into nameElems so they hide along with the rest of
  // the name-entry UI once a name is confirmed.
  buildNameKeyboard() {
    const GAP = 6, KEY_H = 42, topY = 316;
    const KEY_W = Math.floor((W - 32 - 9 * GAP) / 10);

    const addChar = ch => {
      if (this.nameInput.length < 16) { this.nameInput += ch; this.refreshNameDisplay(); }
    };
    const backspace = () => { this.nameInput = this.nameInput.slice(0, -1); this.refreshNameDisplay(); };

    const rows = [
      [...'qwertyuiop'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => addChar(ch) })),
      [...'asdfghjkl'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => addChar(ch) })),
      [...'zxcvbnm'].map(ch => ({ label: ch.toUpperCase(), w: KEY_W, fn: () => addChar(ch) }))
        .concat([{ label: '⌫', w: KEY_W * 1.7, fn: backspace, fill: 0x3a1c1c }]),
      [
        { label: 'SPACE', w: KEY_W * 4.2, fn: () => addChar(' ') },
        { label: '✓ OK',  w: KEY_W * 2.6, fn: () => { if (this.nameInput.length) this.commitName(); }, fill: 0x0a4a2a },
      ],
    ];

    const makeKey = (x, y, spec) => {
      const fill = spec.fill || 0x1c2a3a;
      const bg = this.add.rectangle(x, y, spec.w, KEY_H, fill)
        .setStrokeStyle(1, 0x334455).setInteractive({ useHandCursor: true });
      const txt = this.add.text(x, y, spec.label, {
        fontSize: '16px', fill: '#dde8ff', fontFamily: '"Arial Black", Arial, sans-serif',
      }).setOrigin(0.5);
      this.nameElems.push(bg, txt);
      bg.on('pointerdown', () => { bg.setFillStyle(0x33475e); spec.fn(); });
      bg.on('pointerup',   () => bg.setFillStyle(fill));
      bg.on('pointerout',  () => bg.setFillStyle(fill));
    };

    rows.forEach((row, r) => {
      const y = topY + r * (KEY_H + GAP);
      const rowWidth = row.reduce((s, k) => s + k.w, 0) + (row.length - 1) * GAP;
      let x = (W - rowWidth) / 2 + row[0].w / 2;
      row.forEach((spec, i) => {
        makeKey(x, y, spec);
        if (i < row.length - 1) x += spec.w / 2 + GAP + row[i + 1].w / 2;
      });
    });
  }

  onNameKey(e) {
    if (e.key === 'Enter') {
      if (!this.nameInput.length) return;
      this.commitName(); return;
    }
    if (e.key === 'Backspace') { this.nameInput = this.nameInput.slice(0, -1); this.refreshNameDisplay(); return; }
    if (e.key.length === 1 && /[\x20-\x7e]/.test(e.key) && this.nameInput.length < 16) {
      this.nameInput += e.key; this.refreshNameDisplay();
    }
  }

  refreshNameDisplay() {
    if (this.nameDisplay) this.nameDisplay.setText(this.nameInput + (this.cursorVis ? '_' : ' '));

    }).setOrigin(1, 0.5);

    this.add.text(90,  hY, '#',     hSt).setOrigin(0.5, 0);

  _renderLbRows(fullList, isGlobal) {
    for (const o of this.lbRowElems) o.destroy();
    this.lbRowElems = [];

      .setStyle({ fill: isGlobal ? '#44ffaa' : '#445566' });

    const lb = fullList.slice(0, 10);
    const hY = 207;
    const C  = { rank: 90, name: 118, score: 420, level: 470, mode: 555, date: 700 };

    if (!lb.length) {
      push(this.add.text(W / 2, hY + 36, 'No entries yet — you will be the first!', {
        fontSize: '14px', fill: '#334455', fontFamily: 'Arial, sans-serif',
      }).setOrigin(0.5));
      return;
    }

    const highlightName = this._lbHighlight;
    const medals = ['#ffd700', '#c0c0c0', '#cd7f32'];
    lb.forEach((e, i) => {
      const y     = hY + 24 + i * 22;
      const isNew = highlightName !== null
        && e.name === (this.nameInput || highlightName)
        && e.score === this.finalScore
        && i === lb.findIndex(x => x.name === e.name && x.score === e.score);

      if (isNew) push(this.add.rectangle(W / 2, y + 9, 660, 21, 0x1a3a1a).setAlpha(0.8));
      else if (i < 3) push(this.add.rectangle(W / 2, y + 9, 660, 21, 0x1a2234).setAlpha(0.5));

      const col   = isNew ? '#88ff88' : i < 3 ? medals[i] : '#667788';
      const rank  = i === 0 ? '🥇' : i === 1 ? '🥈' : i === 2 ? '🥉' : `${i + 1}.`;
      const rs    = { fontSize: '13px', fill: col, fontFamily: 'Arial, sans-serif' };
      const lvlBase = e.reachedLevel != null && e.reachedLevel !== e.startLevel
        ? `${e.startLevel}→${e.reachedLevel}` : `${e.startLevel ?? '?'}`;
      const lvl   = e.kid ? `${lvlBase} 👶` : lvlBase;
      const meta  = modeMeta(e.mode);

      push(this.add.text(C.rank,  y, rank,              rs).setOrigin(0.5, 0));
      push(this.add.text(C.name,  y, e.name,            rs).setOrigin(0, 0));
      push(this.add.text(C.score, y, `${e.score}`,      rs).setOrigin(1, 0));
      push(this.add.text(C.level, y, lvl,               rs).setOrigin(0.5, 0));
      push(this.add.text(C.mode,  y, meta.short, { fontSize: '11px', fill: meta.color, fontFamily: 'Arial, sans-serif' }).setOrigin(0.5, 0));
      push(this.add.text(C.date,  y, e.date || '',      { fontSize: '11px', fill: '#2a3a4a', fontFamily: 'Arial, sans-serif' }).setOrigin(1, 0));
      if (isNew) push(this.add.text(C.date + 16, y, '◄ new', { fontSize: '11px', fill: '#44ff88', fontFamily: 'Arial, sans-serif' }).setOrigin(0, 0));
    });
  }

  buildButtons() {
    const pb = makeButton(this, W / 2 - 115, 548, 200, 52, 0x0055aa, 'PLAY AGAIN', 20);
    pb.bg.on('pointerdown', () => this.scene.start('Game', {
      level: this.startLevel, kid: this.kidMode, mode: this.mechanic,
      unit: this.unit, lbMode: this.lbMode, back: this.backTo,
    }));

    const mb = this.add.rectangle(W / 2 + 115, 548, 200, 52, 0x1e2a3a)
      .setStrokeStyle(2, 0x334455).setInteractive({ useHandCursor: true });
    this.add.text(W / 2 + 115, 548, 'MAIN MENU', {
      fontSize: '20px', fill: '#9aaabb', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5);
    mb.on('pointerover', () => mb.setFillStyle(0x273550));
    mb.on('pointerout',  () => mb.setFillStyle(0x1e2a3a));
    mb.on('pointerdown', () => this.scene.start(this.backTo.scene, this.backTo.data));
  }
}

/* ════════════════════════════════════════════════════════════════════════════
   LEADERBOARD SCENE

class LeaderboardScene extends Phaser.Scene {
  constructor() { super('Leaderboard'); }

  create() {
    this.filterMode = 'all'; // 'all' | 'tricky' | 'typing' | 'translation' | 'story'
    this.tableElems = [];
    this.globalData = null; // null until the shared Firebase list arrives (or fails)

    this.add.rectangle(W / 2, H / 2, W, H, 0x0d0d1a);
    drawGrid(this, 0x141428, 0.6);

    this.add.text(W / 2, 38, '🏆  LEADERBOARD', {
      fontSize: '40px', fill: '#ffdd00',
      fontFamily: '"Arial Black", Arial, sans-serif',
      stroke: '#443300', strokeThickness: 6,
    }).setOrigin(0.5);

    this.sourceBadge = this.add.text(W - 16, 18, '📱 local', {
      fontSize: '11px', fill: '#556677', fontFamily: 'Arial, sans-serif',
    }).setOrigin(1, 0.5);

    /* ── Filter tabs: Overall + one per game mode ─────────────────────────── */
    this.tabCfg = [
      { id: 'all',         label: 'OVERALL',                   color: '#dddddd' },
      { id: 'tricky',      label: MODE_META.tricky.label,      color: MODE_META.tricky.color },
      { id: 'typing',      label: MODE_META.typing.label,      color: MODE_META.typing.color },
      { id: 'translation', label: MODE_META.translation.label, color: MODE_META.translation.color },
      { id: 'story',       label: MODE_META.story.label,       color: MODE_META.story.color },
    ];
    const N = this.tabCfg.length;
    const TW = 136, TH = 36, GAP = 6;
    const startX = (W - (N * TW + (N - 1) * GAP)) / 2 + TW / 2;
    this.tabBgs = []; this.tabTxts = [];
    this.tabCfg.forEach((cfg, i) => {
      const x = startX + i * (TW + GAP), y = 74;
      const bg = this.add.rectangle(x, y, TW, TH, 0x141e2a).setStrokeStyle(2, 0x223344).setInteractive({ useHandCursor: true });
      const txt = this.add.text(x, y, cfg.label, {
        fontSize: '12px', fill: cfg.color,
        fontFamily: '"Arial Black", Arial, sans-serif',
      }).setOrigin(0.5);
      this.tabBgs.push(bg); this.tabTxts.push(txt);
      bg.on('pointerover', () => { if (this.filterMode !== cfg.id) bg.setFillStyle(0x1c2838); });
      bg.on('pointerout',  () => { if (this.filterMode !== cfg.id) bg.setFillStyle(0x141e2a); });
      bg.on('pointerdown', () => { this.filterMode = cfg.id; this.refreshTabs(); this.renderTable(); });
    });
    this.refreshTabs();

    this.renderTable();
    fetchGlobalLeaderboard(arr => {
      if (arr) {
        this.globalData = arr;
        this.sourceBadge.setText('🌍 global').setStyle({ fill: '#44ffaa' });
        this.renderTable();
      }
    });

    /* Back button */
    const back = this.add.rectangle(W / 2 - 120, 562, 200, 48, 0x1e2a3a)
      .setStrokeStyle(2, 0x334455).setInteractive({ useHandCursor: true });
    this.add.text(W / 2 - 120, 562, '← BACK', {
      fontSize: '20px', fill: '#9aaabb', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5);
    back.on('pointerover', () => back.setFillStyle(0x273550));
    back.on('pointerout',  () => back.setFillStyle(0x1e2a3a));
    back.on('pointerdown', () => this.scene.start('Menu'));

    /* Clear button (two-click confirm) — clears this device's LOCAL cache only.
       The shared global leaderboard is intentionally not wipeable from here. */
    this.clearConfirm = false;
    const clrBg = this.add.rectangle(W / 2 + 120, 562, 200, 48, 0x2a1a1a)
      .setStrokeStyle(2, 0x442222).setInteractive({ useHandCursor: true });
    this.clrTxt = this.add.text(W / 2 + 120, 562, 'CLEAR LOCAL', {
      fontSize: '18px', fill: '#884444', fontFamily: 'Arial, sans-serif',
    }).setOrigin(0.5);
    clrBg.on('pointerover', () => clrBg.setFillStyle(0x3a2020));
    clrBg.on('pointerout',  () => clrBg.setFillStyle(0x2a1a1a));
    clrBg.on('pointerdown', () => {
      if (!this.clearConfirm) {
        this.clearConfirm = true;
        this.clrTxt.setText('CONFIRM?').setStyle({ fill: '#ff4444' });
      } else {
        saveLeaderboard([]);
        this.scene.restart();
      }
    });
  }

  refreshTabs() {
    this.tabCfg.forEach((cfg, i) => {
      const sel = cfg.id === this.filterMode;
      this.tabBgs[i].setFillStyle(sel ? 0x22334a : 0x141e2a).setStrokeStyle(2, sel ? 0xffffff : 0x223344);
      this.tabTxts[i].setAlpha(sel ? 1 : 0.6);
    });
  }

  /* Rebuilds just the table portion (header + rows) for the current filter,
     without touching the title, tabs, or bottom buttons. */
  renderTable() {
    for (const o of this.tableElems) o.destroy();
    this.tableElems = [];
    const push = o => { this.tableElems.push(o); return o; };

    const all = this.globalData || getLeaderboard();
    const lb  = (this.filterMode === 'all'
      ? all
      : all.filter(e => (e.mode || 'legacy') === this.filterMode)
    ).slice(0, 10);

    const C  = { rank: 90, name: 122, score: 470, level: 528, mode: 610, date: 700 };
    const hY = 106;

    const hSt = { fontSize: '12px', fill: '#445566', fontFamily: 'Arial, sans-serif', letterSpacing: 1 };
    push(this.add.text(C.rank,  hY, '#',      hSt).setOrigin(0.5, 0));
    push(this.add.text(C.name,  hY, 'NAME',   hSt).setOrigin(0, 0));
    push(this.add.text(C.score, hY, 'SCORE',  hSt).setOrigin(1, 0));
    push(this.add.text(C.level, hY, 'LEVEL',  hSt).setOrigin(0.5, 0));
    push(this.add.text(C.mode,  hY, 'MODE',   hSt).setOrigin(0.5, 0));
    push(this.add.text(C.date,  hY, 'DATE',   hSt).setOrigin(1, 0));
    push(this.add.rectangle(W / 2, hY + 19, 662, 1, 0x223344));

    if (!lb.length) {
      const msg = this.filterMode === 'all'
        ? 'No scores yet.\nPlay a game to get on the board!'
        : `No scores yet in "${modeMeta(this.filterMode).label}".\nPlay that mode to be the first!`;
      push(this.add.text(W / 2, 300, msg, {
        fontSize: '22px', fill: '#334455', fontFamily: 'Arial, sans-serif', align: 'center',
      }).setOrigin(0.5));
      return;
    }

    const medals = ['#ffd700', '#c0c0c0', '#cd7f32'];
    lb.forEach((e, i) => {
      const y    = hY + 27 + i * 38;
      const col  = i < 3 ? medals[i] : (i % 2 === 0 ? '#8899aa' : '#778899');
      const rank = i === 0 ? '🥇' : i === 1 ? '🥈' : i === 2 ? '🥉' : `${i + 1}.`;
      const rs      = { fontSize: '18px', fill: col, fontFamily: i < 3 ? '"Arial Black", Arial, sans-serif' : 'Arial, sans-serif' };
      const lvlBase = e.reachedLevel != null && e.reachedLevel !== e.startLevel
        ? `${e.startLevel}→${e.reachedLevel}` : `${e.startLevel ?? '?'}`;
      const lvl     = e.kid ? `${lvlBase} 👶` : lvlBase;
      const meta    = modeMeta(e.mode);

      if (i < 3) push(this.add.rectangle(W / 2, y + 14, 662, 36, 0x1a2234).setAlpha(0.6));

      push(this.add.text(C.rank,  y, rank,                       rs).setOrigin(0.5, 0));
      push(this.add.text(C.name,  y, e.name,                     rs).setOrigin(0, 0));
      push(this.add.text(C.score, y, e.score.toLocaleString(),   rs).setOrigin(1, 0));
      push(this.add.text(C.level, y, lvl, { ...rs, fontSize: '14px' }).setOrigin(0.5, 0));
      push(this.add.text(C.mode,  y, meta.short, {
        fontSize: '13px', fill: meta.color, fontFamily: 'Arial, sans-serif',
      }).setOrigin(0.5, 0));
      push(this.add.text(C.date,  y, e.date || '', {
        fontSize: '12px', fill: '#334455', fontFamily: 'Arial, sans-serif',
      }).setOrigin(1, 0));
    });
  }
}
